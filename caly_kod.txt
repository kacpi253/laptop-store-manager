import javax.swing.*;
import java.awt.*;
import java.util.List;

public class AdminFrame extends BaseStoreFrame {

    public AdminFrame(StoreFacade facade, Runnable onLogout) {
        super("Sklep z laptopami - Panel Administratora", "Administrator", facade, onLogout);
    }

    @Override
    protected void setupActionPanel(JPanel actionPanel) {
        // Układ 2 wiersze x 4 kolumny, żeby 7 przycisków zawsze wyglądało równo
        actionPanel.setLayout(new GridLayout(2, 4, 8, 8));
        actionPanel.setBorder(BorderFactory.createEmptyBorder(8, 12, 10, 12));

        JButton addBranchBtn = new JButton("Dodaj oddział");
        JButton removeBranchBtn = new JButton("Usuń oddział");
        JButton addLaptopBtn = new JButton("Dodaj laptop");
        JButton removeLaptopBtn = new JButton("Usuń laptop");
        JButton editConditionBtn = new JButton("Edytuj stan");
        JButton editPriceBtn = new JButton("Edytuj cenę");
        JButton contactRequestsBtn = new JButton("Prośby o kontakt");

        addBranchBtn.addActionListener(e -> handleAddBranch());
        removeBranchBtn.addActionListener(e -> handleRemoveBranch());
        addLaptopBtn.addActionListener(e -> handleAddLaptop());
        removeLaptopBtn.addActionListener(e -> handleRemoveLaptop());
        editConditionBtn.addActionListener(e -> handleEditCondition());
        editPriceBtn.addActionListener(e -> handleEditPrice());
        contactRequestsBtn.addActionListener(e -> handleShowContactRequests());

        actionPanel.add(addBranchBtn);
        actionPanel.add(removeBranchBtn);
        actionPanel.add(addLaptopBtn);
        actionPanel.add(removeLaptopBtn);
        actionPanel.add(editConditionBtn);
        actionPanel.add(editPriceBtn);
        actionPanel.add(contactRequestsBtn);
    }

    private void handleAddBranch() {
        JTextField nameField = new JTextField();
        JTextField capacityField = new JTextField();

        JPanel form = new JPanel(new GridLayout(0, 2, 5, 5));
        form.add(new JLabel("Nazwa oddziału:"));
        form.add(nameField);
        form.add(new JLabel("Maks. pojemność:"));
        form.add(capacityField);

        int result = JOptionPane.showConfirmDialog(this, form, "Dodaj nowy oddział",
                JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);

        if (result == JOptionPane.OK_OPTION) {
            String name = nameField.getText().trim();
            if (name.isEmpty()) {
                showError("Nazwa oddziału nie może być pusta.");
                return;
            }
            try {
                int capacity = Integer.parseInt(capacityField.getText().trim());
                if (capacity <= 0) {
                    showError("Pojemność musi być liczbą większą od 0.");
                    return;
                }
                facade.addBranch(name, capacity);
                reloadBranches();
            } catch (NumberFormatException ex) {
                showError("Pojemność musi być poprawną liczbą całkowitą.");
            }
        }
    }

    private void handleRemoveBranch() {
        Branch selected = branchTablePanel.getSelectedItem();
        if (selected == null) {
            showError("Zaznacz oddział, który chcesz usunąć.");
            return;
        }
        facade.removeBranch(selected.getBranchName());
        reloadBranches();
    }

    private void handleAddLaptop() {
        Branch selectedBranch = branchTablePanel.getSelectedItem();
        if (selectedBranch == null) {
            showError("Najpierw zaznacz oddział, do którego chcesz dodać laptopa.");
            return;
        }

        JTextField brandField = new JTextField();
        JTextField modelField = new JTextField();
        JComboBox<LaptopCondition> conditionBox = new JComboBox<>(LaptopCondition.values());
        JTextField yearField = new JTextField("2023");
        JTextField priceField = new JTextField("2500.0");
        JTextField batteryField = new JTextField("90.0");
        JTextField osField = new JTextField("Windows 11");
        JTextField storageField = new JTextField("512");

        JPanel form = new JPanel(new GridLayout(0, 2, 5, 5));
        form.add(new JLabel("Marka:"));
        form.add(brandField);
        form.add(new JLabel("Model:"));
        form.add(modelField);
        form.add(new JLabel("Stan:"));
        form.add(conditionBox);
        form.add(new JLabel("Rok produkcji:"));
        form.add(yearField);
        form.add(new JLabel("Cena (PLN):"));
        form.add(priceField);
        form.add(new JLabel("Kondycja baterii (%):"));
        form.add(batteryField);
        form.add(new JLabel("System operacyjny:"));
        form.add(osField);
        form.add(new JLabel("Dysk (GB):"));
        form.add(storageField);

        int result = JOptionPane.showConfirmDialog(this, form,
                "Dodaj laptop do: " + selectedBranch.getBranchName(),
                JOptionPane.OK_CANCEL_OPTION, JOptionPane.PLAIN_MESSAGE);

        if (result == JOptionPane.OK_OPTION) {
            try {
                String brand = brandField.getText().trim();
                String model = modelField.getText().trim();
                LaptopCondition condition = (LaptopCondition) conditionBox.getSelectedItem();
                int year = Integer.parseInt(yearField.getText().trim());
                double price = Double.parseDouble(priceField.getText().trim());
                double battery = Double.parseDouble(batteryField.getText().trim());
                String os = osField.getText().trim();
                int storage = Integer.parseInt(storageField.getText().trim());

                if (brand.isEmpty() || model.isEmpty()) {
                    showError("Marka i model nie mogą być puste.");
                    return;
                }

                // Uwaga na kolejność argumentów Twojego konstruktora Laptop: najpierw model, potem brand
                Laptop newLaptop = new Laptop(model, brand, condition, year, price, battery, os, storage);
                facade.addLaptopToBranch(selectedBranch, newLaptop);

                branchTablePanel.refresh();
                refreshLaptopTable();
            } catch (NumberFormatException ex) {
                showError("Błędny format liczby (sprawdź rok, cenę, baterię lub dysk).");
            } catch (InvalidPriceException | BranchFullException | DuplicateLaptopException | LaptopNotFoundException ex) {
                // Przechwycenie wszystkich naszych własnych wyjątków checked i pokazanie ich w JOptionPane
                showError(ex.getMessage());
            }
        }
    }

    private void handleRemoveLaptop() {
        Branch selectedBranch = branchTablePanel.getSelectedItem();
        Laptop selectedLaptop = laptopTablePanel.getSelectedItem();

        try {
            facade.removeLaptopFromBranch(selectedBranch, selectedLaptop);
            branchTablePanel.refresh();
            refreshLaptopTable();
        } catch (LaptopNotFoundException ex) {
            showError(ex.getMessage());
        }
    }

    private void handleEditCondition() {
        Laptop selectedLaptop = laptopTablePanel.getSelectedItem();
        if (selectedLaptop == null) {
            showError("Zaznacz laptopa, którego stan chcesz zmienić.");
            return;
        }

        LaptopCondition newCondition = (LaptopCondition) JOptionPane.showInputDialog(
                this,
                "Wybierz nowy stan dla: " + selectedLaptop.getBrand() + " " + selectedLaptop.getModel(),
                "Edycja stanu",
                JOptionPane.PLAIN_MESSAGE,
                null,
                LaptopCondition.values(),
                selectedLaptop.getCondition()
        );

        if (newCondition != null) {
            try {
                facade.updateLaptopCondition(selectedLaptop, newCondition);
                refreshLaptopTable();
            } catch (LaptopNotFoundException ex) {
                showError(ex.getMessage());
            }
        }
    }

    private void handleEditPrice() {
        Laptop selectedLaptop = laptopTablePanel.getSelectedItem();
        if (selectedLaptop == null) {
            showError("Zaznacz laptopa, którego cenę chcesz zmienić.");
            return;
        }

        String input = JOptionPane.showInputDialog(
                this,
                "Podaj nową cenę (PLN) dla: " + selectedLaptop.getBrand() + " " + selectedLaptop.getModel(),
                selectedLaptop.getPrice()
        );

        if (input != null) {
            try {
                double newPrice = Double.parseDouble(input.trim());
                facade.updateLaptopPrice(selectedLaptop, newPrice);
                refreshLaptopTable();
            } catch (NumberFormatException ex) {
                showError("Wprowadź poprawną wartość liczbową.");
            } catch (LaptopNotFoundException | InvalidPriceException ex) {
                showError(ex.getMessage());
            }
        }
    }

    private void handleShowContactRequests() {
        List<String> requests = facade.getContactRequests();
        if (requests.isEmpty()) {
            showInfo("Brak nowych próśb o kontakt od klientów.");
            return;
        }

        StringBuilder sb = new StringBuilder("Zgłoszenia od klientów:\n\n");
        for (int i = 0; i < requests.size(); i++) {
            sb.append(i + 1).append(". ").append(requests.get(i)).append("\n");
        }
        showInfo(sb.toString());
    }
}

import javax.swing.*;
import java.awt.*;
import java.util.List;

public abstract class BaseStoreFrame extends JFrame {
    protected final StoreFacade facade;
    protected final GenericTablePanel<Branch> branchTablePanel;
    protected final GenericTablePanel<Laptop> laptopTablePanel;

    private final JTextField filterTextField;
    private final JComboBox<Object> conditionComboBox;
    private final Runnable onLogout;

    public BaseStoreFrame(String title, String roleLabel, StoreFacade facade, Runnable onLogout) {
        super(title);
        this.facade = facade;
        this.onLogout = onLogout;

        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setSize(950, 650);
        setLocationRelativeTo(null); // Wyśrodkowanie okna na ekranie
        setLayout(new BorderLayout(10, 10));

        // 1. GÓRNY PASEK (rola + wylogowanie)
        JPanel headerPanel = new JPanel(new BorderLayout());
        headerPanel.setBorder(BorderFactory.createEmptyBorder(8, 12, 0, 12));
        JLabel userLabel = new JLabel("Zalogowano jako: " + roleLabel);
        userLabel.setFont(userLabel.getFont().deriveFont(Font.BOLD, 14f));

        JButton logoutButton = new JButton("Wyloguj");
        logoutButton.addActionListener(e -> {
            dispose(); // Zamyka bieżące okno
            if (this.onLogout != null) {
                this.onLogout.run(); // Wraca do okna logowania z tą samą Fasadą
            }
        });
        headerPanel.add(userLabel, BorderLayout.WEST);
        headerPanel.add(logoutButton, BorderLayout.EAST);
        add(headerPanel, BorderLayout.NORTH);

        // 2. TABELA ODDZIAŁÓW (nasz generyczny komponent dla typu Branch)
        branchTablePanel = new GenericTablePanel<>(
                "Oddziały sklepu",
                new String[]{"Nazwa oddziału", "Liczba laptopów", "Pojemność", "Zapełnienie (%)"},
                List.of(
                        Branch::getBranchName,
                        Branch::getLaptopCount,
                        Branch::getMaxCapacity,
                        b -> String.format("%.1f%%", b.getCurrentLoad())
                )
        );

        // Przycisk sortowania oddziałów po aktualnym zapełnieniu (przekazanie referencji listy)
        JButton sortBranchesButton = new JButton("Sortuj oddziały wg zapełnienia");
        sortBranchesButton.addActionListener(e -> {
            facade.sortBranchesByLoad(branchTablePanel.getData());
            branchTablePanel.refresh();
        });

        JPanel branchTopSection = new JPanel(new BorderLayout(5, 5));
        branchTopSection.add(branchTablePanel, BorderLayout.CENTER);
        JPanel branchSortBar = new JPanel(new FlowLayout(FlowLayout.LEFT));
        branchSortBar.add(sortBranchesButton);
        branchTopSection.add(branchSortBar, BorderLayout.SOUTH);

        // 3. PASEK FILTRÓW + TABELA LAPTOPÓW (ten sam generyczny komponent dla typu Laptop)
        JPanel filterPanel = new JPanel(new FlowLayout(FlowLayout.LEFT, 10, 5));
        filterPanel.add(new JLabel("Filtr (model/marka + Enter):"));

        filterTextField = new JTextField(18);
        // ActionListener w JTextField uruchamia się po naciśnięciu klawisza Enter
        filterTextField.addActionListener(e -> refreshLaptopTable());
        filterPanel.add(filterTextField);

        filterPanel.add(new JLabel("Stan:"));
        conditionComboBox = new JComboBox<>();
        conditionComboBox.addItem("Wszystkie"); // Pierwsza opcja = brak filtra stanu (null)
        for (LaptopCondition condition : LaptopCondition.values()) {
            conditionComboBox.addItem(condition); // JComboBox sam wywoła Twoje toString()
        }
        conditionComboBox.addActionListener(e -> refreshLaptopTable());
        filterPanel.add(conditionComboBox);

        laptopTablePanel = new GenericTablePanel<>(
                "Laptopy w wybranym oddziale",
                new String[]{"Marka", "Model", "Stan", "Rok", "Cena (PLN)", "Bateria (%)", "System", "Dysk (GB)"},
                List.of(
                        Laptop::getBrand,
                        Laptop::getModel,
                        Laptop::getCondition,
                        Laptop::getYear,
                        Laptop::getPrice,
                        Laptop::getBatteryHealth,
                        Laptop::getOperatingSystem,
                        Laptop::getStorageGB
                )
        );

        JPanel laptopBottomSection = new JPanel(new BorderLayout(5, 5));
        laptopBottomSection.add(filterPanel, BorderLayout.NORTH);
        laptopBottomSection.add(laptopTablePanel, BorderLayout.CENTER);

        // Połączenie obu sekcji dzielnikiem (JSplitPane)
        JSplitPane splitPane = new JSplitPane(JSplitPane.VERTICAL_SPLIT, branchTopSection, laptopBottomSection);
        splitPane.setDividerLocation(210);
        splitPane.setResizeWeight(0.35);
        add(splitPane, BorderLayout.CENTER);

        // 4. REAKCJA NA KLIKNIĘCIE ODDZIAŁU W GÓRNEJ TABELI
        branchTablePanel.getTable().getSelectionModel().addListSelectionListener(e -> {
            if (!e.getValueIsAdjusting()) {
                refreshLaptopTable();
            }
        });

        // 5. DOLNY PANEL AKCJI (każda rola wypełni go po swojemu)
        JPanel actionPanel = new JPanel(new FlowLayout(FlowLayout.CENTER, 10, 10));
        setupActionPanel(actionPanel);
        add(actionPanel, BorderLayout.SOUTH);

        // Załadowanie początkowych danych oddziałów z Fasady
        reloadBranches();
    }

    // Metoda abstrakcyjna – klasy potomne (AdminFrame i ClientFrame) dodają tu swoje przyciski
    protected abstract void setupActionPanel(JPanel actionPanel);

    protected void reloadBranches() {
        branchTablePanel.setData(facade.getAllBranches());
        refreshLaptopTable();
    }

    protected void refreshLaptopTable() {
        Branch selectedBranch = branchTablePanel.getSelectedItem();
        String query = filterTextField.getText();

        Object selectedOption = conditionComboBox.getSelectedItem();
        // Jeśli wybrano "Wszystkie" (String), przekazujemy null; jeśli enum, rzutujemy na LaptopCondition
        LaptopCondition condition = (selectedOption instanceof LaptopCondition)
                ? (LaptopCondition) selectedOption
                : null;

        List<Laptop> filtered = facade.filterLaptops(selectedBranch, query, condition);
        laptopTablePanel.setData(filtered);
    }

    // Pomocnicze okienka dialogowe do obsługi wyjątków i komunikatów
    protected void showError(String message) {
        JOptionPane.showMessageDialog(this, message, "Błąd", JOptionPane.ERROR_MESSAGE);
    }

    protected void showInfo(String message) {
        JOptionPane.showMessageDialog(this, message, "Informacja", JOptionPane.INFORMATION_MESSAGE);
    }
}

import java.util.*;

public class Branch {
    private String branchName;
    private List<Laptop> laptopList;
    private int maxCapacity;

    public Branch(String branchName, int maxCapacity){
        this.branchName = branchName;
        this.laptopList = new ArrayList<>();
        this.maxCapacity = maxCapacity;
    }

    public void addLaptop(Laptop laptop){
        if (laptopList.size() >= maxCapacity){
            throw new BranchFullException("Przekrocozno liczbę laptopów");
        }
        for (Laptop l : laptopList ){
            if (l.compareTo(laptop) == 0){
                throw new DuplicateLaptopException("Podany laptop istnieje już w bazie oddziału!");
            }
        }
        laptopList.add(laptop);
    }

    public void removeLaptop(Laptop laptop) {

        boolean removed = laptopList.removeIf(l -> l.compareTo(laptop) == 0);
        if (!removed) {
            throw new LaptopNotFoundException("Nie znaleziono laptopa: " + laptop.getModel());
        }
    }

    public void sellLaptop(Laptop laptop){
        boolean sold= laptopList.removeIf(s -> s.compareTo(laptop) == 0);
        if (!sold) {
            throw new LaptopNotFoundException("Nie znaleziono laptopa do sprzedaży: " + laptop.getModel());
        }
    }

    public void changeCondition(Laptop laptop, LaptopCondition condition){
        laptop.setCondition(condition);
        System.out.println("Zmieniono kondycję laptopa na: " + condition.toString());
    }

    public void changeYear(Laptop laptop, int year){
        laptop.setYear(year);
        System.out.println("Zmieniono rok produkcji laptopa na: " + year);
    }

    public int countByCondition(LaptopCondition condition){
        int c = 0;
        for (Laptop l : laptopList){
            if(l.getCondition() == condition){
                c++;
            }
        }
        return c;
    }

    public List<Laptop> sortByName(){
        List<Laptop> sortedByName = new ArrayList<>(laptopList);
        Collections.sort(sortedByName);
        return sortedByName;
    }

    public List<Laptop> sortByPrice(){
        List<Laptop> sortedByPrice = new ArrayList<>(laptopList);
        sortedByPrice.sort(Comparator.comparingDouble(Laptop::getPrice));
        return sortedByPrice;
    }

    public List<Laptop> search(String model){
        List<Laptop> names = new ArrayList<>();
        for (Laptop l : laptopList){
            if (l.getModel().equalsIgnoreCase(model)){
                names.add(l);
            }
        }

        return names;
    }

    public List<Laptop> searchPartial(String text){
        List<Laptop> results = new ArrayList<>();
        for (Laptop l : laptopList){
            if (l.getModel().toLowerCase().contains(text.toLowerCase()) ||
                    l.getBrand().toLowerCase().contains(text.toLowerCase())){
                results.add(l);
            }
        }

        return results;

    }

    public void summary(){
        System.out.println("Oto zestawienie wszystkich laptopów: ");
        for (Laptop l : laptopList){
            l.print();
        }
    }

    public Laptop max(){
        if (laptopList.isEmpty()) {
            return null;
        }
        return Collections.max(laptopList, Comparator.comparingDouble(Laptop::getPrice));
    }

    public double totalValue() {
        double total = 0.0;
        for (Laptop l : laptopList) {
            total += l.getPrice();
        }
        return total;
    }

    public int countByBrand(String brand) {
        int count = 0;
        for (Laptop l : laptopList) {
            if (l.getBrand().equalsIgnoreCase(brand)) {
                count++;
            }
        }
        return count;
    }

    public void transferOldestOfTwoExpensive(Branch targetBranch) throws LaptopNotFoundException, BranchFullException, DuplicateLaptopException {
        if(targetBranch == null) {
            throw new LaptopNotFoundException("Podany oddział nie istnieje.");
        }
        if (laptopList.size() < 2) {
            throw new LaptopNotFoundException("Oddział ma mniej niż 2 laptopy, brak możliwości wyboru.");
        }
        if (targetBranch.getLaptopCount() >= targetBranch.getMaxCapacity()) {
            throw new BranchFullException("Oddział docelowy nie ma wolnego miejsca.");
        }
        List<Laptop> sorted = sortByPrice();

        Laptop top1 = sorted.get(sorted.size() - 1);
        Laptop top2 = sorted.get(sorted.size() - 2);
        Laptop chosen = (top1.getYear() <= top2.getYear() ? top1 : top2);

        targetBranch.addLaptop(chosen);
        this.removeLaptop(chosen);

        System.out.println("Przeniesiono laptop: " + chosen.getModel() + " (" + chosen.getYear() + ") do oddziału " + targetBranch.getBranchName());

    }
    public String getBranchName() {
        return branchName;
    }

    public int getLaptopCount() {
        return laptopList.size();
    }

    public int getMaxCapacity() {
        return maxCapacity;
    }

    public boolean isEmpty() {
        return laptopList.isEmpty();
    }

    public List<Laptop> getLaptopList() {
        return laptopList;
    }

    public double getCurrentLoad() {
        if (maxCapacity == 0) return 0.0;
        return (double) laptopList.size() / maxCapacity * 100.0;
    }

}

import javax.swing.*;

public class ClientFrame extends BaseStoreFrame {

    public ClientFrame(StoreFacade facade, Runnable onLogout) {
        super("Sklep z laptopami - Panel Klienta", "Klient", facade, onLogout);
    }

    @Override
    protected void setupActionPanel(JPanel actionPanel) {
        JButton reserveButton = new JButton("Zarezerwuj laptop");
        JButton contactButton = new JButton("Poproś o kontakt");

        reserveButton.addActionListener(e -> handleReserveLaptop());
        contactButton.addActionListener(e -> handleContactRequest());

        actionPanel.add(reserveButton);
        actionPanel.add(contactButton);
    }

    private void handleReserveLaptop() {
        Laptop selectedLaptop = laptopTablePanel.getSelectedItem();
        if (selectedLaptop != null && selectedLaptop.getCondition() == LaptopCondition.ZAREZERWOWANY) {
            showInfo("Ten laptop jest już zarezerwowany.");
            return;
        }

        try {
            facade.reserveLaptop(selectedLaptop);
            refreshLaptopTable();
            showInfo("Laptop " + selectedLaptop.getBrand() + " " + selectedLaptop.getModel() + " został zarezerwowany!");
        } catch (LaptopNotFoundException ex) {
            // Obsługa naszego własnego wyjątku za pomocą JOptionPane
            showError(ex.getMessage());
        }
    }

    private void handleContactRequest() {
        Branch selectedBranch = branchTablePanel.getSelectedItem();
        Laptop selectedLaptop = laptopTablePanel.getSelectedItem();

        if (selectedBranch == null || selectedLaptop == null) {
            showError("Najpierw wybierz oddział oraz konkretny laptop z tabeli.");
            return;
        }

        String clientContact = JOptionPane.showInputDialog(
                this,
                "Podaj swój telefon lub e-mail w sprawie: " + selectedLaptop.getBrand() + " " + selectedLaptop.getModel(),
                "Prośba o kontakt",
                JOptionPane.QUESTION_MESSAGE
        );

        // Jeśli użytkownik kliknął Anuluj (null) albo zostawił puste pole, nic nie zapisujemy
        if (clientContact != null && !clientContact.trim().isEmpty()) {
            facade.addContactRequest(selectedBranch, selectedLaptop, clientContact.trim());
            showInfo("Twoja prośba o kontakt została wysłana do administratora!");
        }
    }
}

import javax.swing.table.AbstractTableModel;
import java.util.ArrayList;
import java.util.List;
import java.util.function.Function;

public class GenericTableModel<T> extends AbstractTableModel {
    private List<T> data;
    private final String[] columnNames;
    private final List<Function<T, Object>> columnExtractors;

    public GenericTableModel(String[] columnNames, List<Function<T, Object>> columnExtractors) {
        this.data = new ArrayList<>();
        this.columnNames = columnNames;
        this.columnExtractors = columnExtractors;
    }

    @Override
    public int getRowCount() {
        return data.size();
    }

    @Override
    public int getColumnCount() {
        return columnNames.length;
    }

    @Override
    public String getColumnName(int column) {
        return columnNames[column];
    }

    @Override
    public Object getValueAt(int rowIndex, int columnIndex) {
        // 1. Wyciągamy obiekt z danego wiersza (np. konkretny Laptop albo Branch)
        T item = data.get(rowIndex);
        // 2. Uruchamiamy funkcję przypisaną do tej kolumny (np. Laptop::getPrice)
        return columnExtractors.get(columnIndex).apply(item);
    }

    // Podmienia dane w tabeli i powiadamia JTable, że ma się przerysować
    public void setData(List<T> newData) {
        this.data = (newData != null) ? newData : new ArrayList<>();
        fireTableDataChanged();
    }

    // Zwraca referencję do aktualnej listy w tabeli (kluczowe do sortowania w miejscu!)
    public List<T> getData() {
        return data;
    }

    // Zwraca cały obiekt T z wybranego wiersza (np. po kliknięciu w tabeli)
    public T getItemAt(int rowIndex) {
        if (rowIndex < 0 || rowIndex >= data.size()) {
            return null;
        }
        return data.get(rowIndex);
    }

    // Ręczne odświeżenie widoku po zmianie np. kolejności na liście lub stanu obiektu
    public void refresh() {
        fireTableDataChanged();
    }
}

public class GenericTablePanel<T> extends JPanel {
    private final GenericTableModel<T> tableModel;
    private final JTable table;

    public GenericTablePanel(String title, String[] columnNames, List<Function<T, Object>> extractors) {
        // BorderLayout rozciąga główny komponent (CENTER) na całą dostępną przestrzeń
        super(new BorderLayout());

        // Estetyczna ramka z tytułem tabeli
        setBorder(BorderFactory.createTitledBorder(title));

        this.tableModel = new GenericTableModel<>(columnNames, extractors);
        this.table = new JTable(tableModel);

        // Pozwalamy zaznaczać jeden wiersz naraz (wygodniejsze przy usuwaniu i podglądzie oddziału)
        this.table.setSelectionMode(ListSelectionModel.SINGLE_SELECTION);
        this.table.setFillsViewportHeight(true);

        // JScrollPane - bez niego nie wyświetlają się nagłówki kolumn i paski przewijania
        JScrollPane scrollPane = new JScrollPane(table);
        add(scrollPane, BorderLayout.CENTER);
    }

    public T getSelectedItem() {
        int selectedRow = table.getSelectedRow();
        if (selectedRow == -1) {
            return null;
        }
        int modelRow = table.convertRowIndexToModel(selectedRow);
        return tableModel.getItemAt(modelRow);
    }

    public void setData(List<T> data) {
        tableModel.setData(data);
    }

    public List<T> getData() {
        return tableModel.getData();
    }

    public void refresh() {
        tableModel.refresh();
    }

    public JTable getTable() {
        return table;
    }

import java.util.ArrayList;
import java.util.HashMap;

public class StoreManager {
    private Map<String, Branch> branches;

    public StoreManager(){
         this.branches = new HashMap<>();
    }

    public void addBranch(String name, int capacity){
        branches.put(name, new Branch(name,capacity));
    }

    public void removeBranch(String name){
        branches.remove(name);
    }

    public List<Branch> findEmpty() {
        List<Branch> emptyBranches = new ArrayList<>();
        for (Branch b : branches.values()) {
            if (b.isEmpty()) {
                emptyBranches.add(b);
            }
        }
        return emptyBranches;
    }

    public void summary(){
        System.out.println("Zestawienie oddziałów:");

        for (Branch b : branches.values()) {
            double percent = 0.0;
            if(b.getMaxCapacity() > 0){
                percent = ((double) b.getLaptopCount() / b.getMaxCapacity() * 100.0);
                System.out.printf("Oddział: %s | Zapełnienie: %.2f%%%n", b.getBranchName(), percent);
            }
        }
    }

    public Branch getBranch(String name) {

        return branches.get(name);
    }

    public void transferLaptop(Laptop laptop, String fromBranch, String toBranch)
            throws LaptopNotFoundException, BranchFullException, DuplicateLaptopException {
        Branch from = branches.get(fromBranch);
        Branch to = branches.get(toBranch);

        if (from == null || to == null){
            throw new LaptopNotFoundException("Oddział źródłowy lub docelowy nie istnieje.");
        }
        from.removeLaptop(laptop);
        try {
            to.addLaptop(laptop);
        } catch (BranchFullException | DuplicateLaptopException e){
            from.addLaptop(laptop); // Cofamy usunięcie, jeśli dodanie do celu się nie udało
            throw e;
        }
    }

    public double totalValue(String branchName) {
        Branch b = branches.get(branchName);
        return (b != null) ? b.totalValue() : 0;
    }

    public int countByBrand(String brand) {
        int count = 0;
        for (Branch b : branches.values()) {
            count += b.countByBrand(brand);
        }
        return count;
    }
    public List<Branch> getAllBranches() {
        return new ArrayList<>(branches.values());
    }

}
